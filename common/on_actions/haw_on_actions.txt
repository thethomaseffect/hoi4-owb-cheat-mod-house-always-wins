on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { is_ai = no }
				haw_ensure_player_cheats = yes
				set_country_flag = haw_player_setup
				if = {
					limit = {
						has_idea = haw_instant_intel
						NOT = { has_country_flag = haw_intel_cheated }
					}
					remove_ideas = haw_instant_intel
				}
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				haw_ensure_house_staff = yes
				haw_grant_gated_cheats = yes
			}
		}
	}

	on_research_finished = {
		effect = {
			if = {
				limit = { is_ai = no }
				haw_grant_gated_cheats = yes
			}
		}
	}
}
