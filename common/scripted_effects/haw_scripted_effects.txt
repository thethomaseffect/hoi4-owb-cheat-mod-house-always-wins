haw_finish_scavenging_prep = {
	if = {
		limit = {
			NOT = { has_country_flag = scavenging_enabled }
			NOT = { has_country_flag = scavenging_disabled }
		}
		remove_decision = begin_scavenging_program_wtt
		country_event = scavenge.15
	}
	else = {
		country_event = haw.6
	}
}

haw_complete_scavenging_expedition = {
	if = {
		limit = {
			OR = {
				has_active_mission = scavenging_mission_automated
				has_country_flag = scavenging_underway
			}
		}
		# Always succeed. Fire OWB's real success event so the reward popup shows.
		if = {
			limit = { has_country_flag = giving_scavengers_pa }
			country_event = scavenge.9
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_vehicles }
			country_event = scavenge.5
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_robots }
			country_event = scavenge.17
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_aircraft }
			country_event = scavenge.20
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_beasts }
			country_event = scavenge.23
		}
		else = {
			country_event = scavenge.1
		}
		clr_country_flag = scavenging_underway
		remove_mission = scavenging_mission_automated
	}
	else = {
		country_event = haw.6
	}
}
