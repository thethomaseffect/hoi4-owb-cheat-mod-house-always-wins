	}
}

haw_complete_scavenging_expedition = {
	if = {
		limit = {
			OR = {
				has_active_mission = scavenging_mission_automated
				has_country_flag = scavenging_underway
			}
		}
		# Always succeed. Fire OWB's real success event so the reward popup shows.
		if = {
			limit = { has_country_flag = giving_scavengers_pa }
			country_event = scavenge.9
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_vehicles }
			country_event = scavenge.5
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_robots }
			country_event = scavenge.17
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_aircraft }
			country_event = scavenge.20
		}
		else_if = {
			limit = { has_country_flag = giving_scavengers_beasts }
			country_event = scavenge.23
		}
		else = {
			country_event = scavenge.1
		}
		clr_country_flag = scavenging_underway
		remove_mission = scavenging_mission_automated
	}
	else = {
		country_event = haw.6
	}
}

haw_ncr_election_dump_state = {
	if = {
		limit = { check_variable = { haw_ncr_cand = global.first_candidate_id } }
		set_variable = { global.first_candidate_support_array^ncr_state_id = 100 }
		set_variable = { global.second_candidate_support_array^ncr_state_id = 0 }
	}
	else_if = {
		limit = { check_variable = { haw_ncr_cand = global.second_candidate_id } }
		set_variable = { global.first_candidate_support_array^ncr_state_id = 0 }
		set_variable = { global.second_candidate_support_array^ncr_state_id = 100 }
	}
}

haw_setup_intel = {
	if = {
		limit = { has_dlc = "La Resistance" }
		set_country_flag = haw_intel_cheated
		if = {
			limit = { NOT = { has_idea = haw_instant_intel } }
			add_ideas = haw_instant_intel
		}
		if = {
			limit = { NOT = { has_intelligence_agency = yes } }
			create_intelligence_agency = yes
		}
		if = {
			limit = { NOT = { has_country_flag = haw_ops_spawned } }
			haw_spawn_house_operatives = yes
			set_country_flag = haw_ops_spawned
		}
		if = {
			limit = { NOT = { has_country_flag = haw_agency_maxed } }
			haw_upgrade_house_agency = yes
			set_country_flag = haw_agency_maxed
		}
	}
}
