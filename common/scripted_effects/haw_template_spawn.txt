haw_add_servebot_templates = {
	if = {
		limit = { NOT = { has_tech = haw_servebot_gear_unlock } }
		set_technology = {
			haw_servebot_gear_unlock = 1
			popup = no
		}
	}
	set_technology = {
		haw_servebot_infantry_unlock = 1
		haw_servebot_mobile_unlock = 1
		haw_servebot_robots_unlock = 1
		popup = no
	}
	haw_add_servebot_sf_template = yes
	haw_add_servebot_infantry_template = yes
	haw_add_servebot_mobile_template = yes
	haw_add_servebot_robots_template = yes
	country_event = haw.16
}
